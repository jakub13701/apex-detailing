        function updateSliderPosition(xPos) {

            if (x < 0) x = 0;

        function calculateTotal() {

            const checkboxes = document.querySelectorAll('.calc-checkbox');

            let baseTotal = 0;

            let totalHours = 0;

            let selectedCount = 0;

            let selectedNames = [];


            checkboxes.forEach(cb => {

                if (cb.checked) {

                    baseTotal += parseFloat(cb.value);

                    totalHours += parseInt(cb.getAttribute('data-hours') || 0);

                    selectedCount++;

                    selectedNames.push(cb.getAttribute('data-name'));

                }

            });


            const finalPrice = Math.round(baseTotal * currentMultiplier);


            document.getElementById('totalPriceDisplay').innerText = finalPrice.toLocaleString('cs-CZ') + ' Kč';

            document.getElementById('summaryCount').innerText = selectedCount + ' ' + (selectedCount === 1 ? 'služba' : 'služby');

            document.getElementById('summaryHours').innerText = `cca ${totalHours} hod (${Math.ceil(totalHours/8)} dny)`;


            // Update booking form note with selected preset

            const noteArea = document.getElementById('formServicesNote');

            if (noteArea) {

                noteArea.value = `Kategorie: ${selectedCarTypeName} | Vybrané služby: ${selectedNames.join(', ')} | Odhadovaná cena: ${finalPrice.toLocaleString('cs-CZ')} Kč`;

            }

        }
